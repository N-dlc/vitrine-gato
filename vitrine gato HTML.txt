  <script>
    (function () {
      'use strict';

      /* ---------- Menu mobile ---------- */
      var toggle = document.querySelector('.nav-toggle');
      var menu = document.getElementById('menu-principal');
      function setMenu(open) {
        toggle.setAttribute('aria-expanded', String(open));
        menu.classList.toggle('is-open', open);
      }
      toggle.addEventListener('click', function () {
        setMenu(toggle.getAttribute('aria-expanded') !== 'true');
      });
      menu.addEventListener('click', function (e) {
        if (e.target.closest('a')) { setMenu(false); }
      });
      document.addEventListener('keydown', function (e) {
        if (e.key === 'Escape' && toggle.getAttribute('aria-expanded') === 'true') {
          setMenu(false);
          toggle.focus();
        }
      });

      /* ---------- Apparition discrète + lien actif (si IntersectionObserver existe) ---------- */
      var reveals = document.querySelectorAll('.reveal');
      if ('IntersectionObserver' in window) {
        var revealObs = new IntersectionObserver(function (entries) {
          entries.forEach(function (en) {
            if (en.isIntersecting) { en.target.classList.add('is-visible'); revealObs.unobserve(en.target); }
          });
        }, { threshold: 0.12 });
        reveals.forEach(function (el) { revealObs.observe(el); });

        var links = document.querySelectorAll('.nav-link');
        var secObs = new IntersectionObserver(function (entries) {
          entries.forEach(function (en) {
            if (en.isIntersecting) {
              links.forEach(function (l) {
                if (l.getAttribute('href') === '#' + en.target.id) { l.setAttribute('aria-current', 'true'); }
                else { l.removeAttribute('aria-current'); }
              });
            }
          });
        }, { rootMargin: '-45% 0px -50% 0px' });
        document.querySelectorAll('main section[id]').forEach(function (s) { secObs.observe(s); });
      } else {
        reveals.forEach(function (el) { el.classList.add('is-visible'); });
      }

      /* ---------- Formulaire de réservation (mode brouillon, aucun envoi) ---------- */
      var form = document.getElementById('form-reservation');
      var resultat = document.getElementById('reservation-resultat');
      var erreurGlobale = document.getElementById('erreur-globale');
      var DESTINATAIRE = 'reservation@gato-demo.invalid'; // adresse FICTIVE, non opérationnelle

      // Horaires fictifs en minutes depuis minuit, par jour (0 = dimanche). null = fermé.
      // À garder cohérent avec le tableau de la section Horaires.
      var HORAIRES = {
        0: [450, 780], 1: null, 2: [420, 1140], 3: [420, 1140],
        4: [420, 1140], 5: [420, 1140], 6: [420, 1170]
      };
      var JOURS = ['dimanche', 'lundi', 'mardi', 'mercredi', 'jeudi', 'vendredi', 'samedi'];

      function pad(n) { return (n < 10 ? '0' : '') + n; }
      function hhmm(min) { return pad(Math.floor(min / 60)) + 'h' + pad(min % 60); }
      function aujourdhui() {
        var d = new Date();
        return d.getFullYear() + '-' + pad(d.getMonth() + 1) + '-' + pad(d.getDate());
      }
      function jourDe(iso) {
        var p = iso.split('-');
        return new Date(+p[0], +p[1] - 1, +p[2]).getDay();
      }

      var dateInput = document.getElementById('date');
      dateInput.min = aujourdhui();

      function val(id) { return document.getElementById(id).value.trim(); }

      // Chaque règle renvoie un message d'erreur, ou '' si le champ est valide.
      var regles = {
        nom: function () {
          var v = val('nom');
          if (!v) { return 'Indiquez votre nom et prénom.'; }
          if (v.length < 3) { return 'Le nom semble trop court.'; }
          return '';
        },
        telephone: function () {
          var v = val('telephone');
          if (!v) { return 'Indiquez un numéro de téléphone.'; }
          if (!/^(?:(?:\+|00)33|0)\s?[1-9](?:[\s.\-]?\d{2}){4}$/.test(v)) {
            return 'Numéro invalide. Exemple : 06 12 34 56 78.';
          }
          return '';
        },
        email: function () {
          var v = val('email');
          if (v && !/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v)) { return 'Adresse e-mail invalide. Exemple : nom@exemple.fr.'; }
          return '';
        },
        date: function () {
          var v = val('date');
          if (!v) { return 'Choisissez une date.'; }
          if (v < aujourdhui()) { return 'La date ne peut pas être dans le passé.'; }
          if (!HORAIRES[jourDe(v)]) { return 'La boulangerie est fermée ce jour-là (' + JOURS[jourDe(v)] + '). Choisissez une autre date.'; }
          return '';
        },
        heure: function () {
          var v = val('heure');
          if (!v) { return 'Choisissez une heure de retrait.'; }
          var d = val('date');
          if (d && HORAIRES[jourDe(d)]) {
            var h = HORAIRES[jourDe(d)];
            var p = v.split(':');
            var min = +p[0] * 60 + +p[1];
            if (min < h[0] || min > h[1] - 15) {
              return 'Ce jour-là, le retrait est possible de ' + hhmm(h[0]) + ' à ' + hhmm(h[1] - 15) + '.';
            }
          }
          return '';
        },
        produit: function () {
          return val('produit') ? '' : 'Indiquez le produit ou la commande souhaitée.';
        },
        quantite: function () {
          var v = val('quantite');
          if (!v) { return ''; }
          var n = Number(v);
          if (!Number.isInteger(n) || n < 1 || n > 50) { return 'Indiquez une quantité entière entre 1 et 50.'; }
          return '';
        },
        consentement: function () {
          return document.getElementById('consentement').checked ? '' : 'Cochez la case pour continuer.';
        }
      };
      var ordre = ['nom', 'telephone', 'email', 'date', 'heure', 'produit', 'quantite', 'consentement'];

      function afficherErreur(id, msg) {
        var champ = document.getElementById(id);
        var zone = document.getElementById('err-' + id);
        zone.textContent = msg;
        if (msg) { champ.setAttribute('aria-invalid', 'true'); }
        else { champ.removeAttribute('aria-invalid'); }
      }

      // Efface l'erreur d'un champ dès que l'utilisateur le corrige ; revalide l'heure si la date change.
      ordre.forEach(function (id) {
        var champ = document.getElementById(id);
        var evt = (champ.type === 'checkbox') ? 'change' : 'input';
        champ.addEventListener(evt, function () {
          if (champ.getAttribute('aria-invalid') === 'true') { afficherErreur(id, regles[id]()); }
        });
      });
      dateInput.addEventListener('change', function () {
        if (document.getElementById('heure').getAttribute('aria-invalid') === 'true') {
          afficherErreur('heure', regles.heure());
        }
      });

      function formaterDate(iso) {
        var p = iso.split('-');
        return JOURS[jourDe(iso)] + ' ' + p[2] + '/' + p[1] + '/' + p[0];
      }

      form.addEventListener('submit', function (e) {
        e.preventDefault();
        resultat.hidden = true;
        var premiere = null;
        var nbErreurs = 0;

        ordre.forEach(function (id) {
          var msg = regles[id]();
          afficherErreur(id, msg);
          if (msg) { nbErreurs++; if (!premiere) { premiere = id; } }
        });

        if (nbErreurs) {
          erreurGlobale.textContent = nbErreurs === 1
            ? 'Un champ est à corriger avant de continuer.'
            : nbErreurs + ' champs sont à corriger avant de continuer.';
          erreurGlobale.hidden = false;
          document.getElementById(premiere).focus(); // les valeurs saisies sont conservées
          return;
        }
        erreurGlobale.hidden = true;

        var lignes = [
          ['Nom', val('nom')],
          ['Téléphone', val('telephone')],
          ['E-mail', val('email') || 'non renseigné'],
          ['Date souhaitée', formaterDate(val('date'))],
          ['Heure de retrait', val('heure').replace(':', 'h')],
          ['Commande', val('produit')],
          ['Quantité', val('quantite') || 'non précisée'],
          ['Message', val('message') || 'aucun']
        ];

        // Récapitulatif affiché (construit avec textContent : aucun HTML injecté)
        var recap = document.getElementById('recap');
        recap.textContent = '';
        lignes.forEach(function (l) {
          var dt = document.createElement('dt'); dt.textContent = l[0];
          var dd = document.createElement('dd'); dd.textContent = l[1];
          recap.appendChild(dt); recap.appendChild(dd);
        });

        // Brouillon d'e-mail : s'ouvre uniquement si l'utilisateur clique sur le lien
        var corps = 'Bonjour,\r\n\r\nJe souhaite faire la demande de réservation suivante :\r\n\r\n' +
          lignes.map(function (l) { return l[0] + ' : ' + l[1]; }).join('\r\n') +
          '\r\n\r\nMerci de me confirmer la disponibilité.';
        document.getElementById('lien-mailto').href = 'mailto:' + DESTINATAIRE +
          '?subject=' + encodeURIComponent('Demande de réservation (démonstration)') +
          '&body=' + encodeURIComponent(corps);

        resultat.hidden = false;
        resultat.focus();
        resultat.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
      });
    })();
